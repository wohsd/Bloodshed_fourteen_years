# 计算派系影响力并更新修正效果
CHI_update_faction_effects = {
   
    # 确保效果变量存在
    if = {
        limit = { NOT = { has_variable = CHI_he_army_strength } }
        set_variable = { CHI_he_army_strength = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_chen_army_exp } }
        set_variable = { CHI_chen_army_exp = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_hu_political_power } }
        set_variable = { CHI_hu_political_power = 0.15 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_tang_consumer_goods } }
        set_variable = { CHI_tang_consumer_goods = -0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_tang_consumer_goods_display } }
        set_variable = { CHI_tang_consumer_goods_display = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_hu_power_fuxing } }
        set_variable = { CHI_hu_power_fuxing = 0 }
    }
    
    # 直接设置技能影响力为相应将领的等级（只计算陆军将领）
    set_variable = { CHI_he_power_skill = 0 }
    set_variable = { CHI_chen_power_skill = 0 }
    set_variable = { CHI_hu_power_skill = 0 }
    set_variable = { CHI_tang_power_skill = 0 }
    
    # 计算陆军将领的技能影响力
    every_army_leader = {
        limit = { AND = { has_trait = heyingqinxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_he_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = chenchengxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_chen_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = huzongnanxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_hu_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = tangenboxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_tang_power_skill = skill }
    }
    
    # 基础影响力：雇佣任意拥有对应派系特质的顾问时 +5（通用检测，每日重算，幂等）
    # 派系特质（heyingqinxi/chenchengxi/huzongnanxi/tangenboxi）在角色的军长角色上
    # is_hired_as_advisor = 正被雇佣为顾问（is_advisor 只判断拥有顾问角色，未雇佣也算，不可用）
    set_variable = { CHI_he_power_base = 0 }
    set_variable = { CHI_chen_power_base = 0 }
    set_variable = { CHI_hu_power_base = 0 }
    set_variable = { CHI_tang_power_base = 0 }
    every_character = {

        add_to_variable = { PREV.CHI_he_power_base = 5 }
    }
    every_character = {
        limit = { is_hired_as_advisor = yes has_trait = chenchengxi }
        add_to_variable = { PREV.CHI_chen_power_base = 5 }
    }
    every_character = {
        limit = { is_hired_as_advisor = yes has_trait = huzongnanxi }
        add_to_variable = { PREV.CHI_hu_power_base = 5 }
    }
    every_character = {
        limit = { is_hired_as_advisor = yes has_trait = tangenboxi }
        add_to_variable = { PREV.CHI_tang_power_base = 5 }
    }

    add_to_variable = { CHI_total_clique_power = CHI_chen_power }

        limit = { NOT = { has_variable = CHI_total_clique_power } }

    if = {

        divide_variable = { CHI_tang_power_percent = CHI_total_clique_power }
        multiply_variable = { CHI_tang_power_percent = 100 }
    }
    else {
        set_variable = { CHI_he_power_percent = 25 }
        set_variable = { CHI_chen_power_percent = 25 }
        set_variable = { CHI_hu_power_percent = 25 }
        set_variable = { CHI_tang_power_percent = 25 }
    }
    
    # 应用派系影响力效果
    # 直接根据影响力百分比计算变量值（仅保留正面属性）

    # 何应钦派系

    # 政治点数因子: 0.3 * (影响力百分比/100)
    set_variable = { CHI_hu_political_power = 0.3 }
    multiply_variable = { CHI_hu_political_power = CHI_hu_power_percent }

}

# 领袖权威决议：五大派系影响力系统（CC系/黄埔系/复兴社/财政系/政学系）
# 影响力为百分比制（0-100），除复兴社外，其余四派通过扶植决议分配空间
# 复兴社影响力与法西斯主义支持率绑定（支持率为0则影响力为0）
# 设计为幂等：每次调用先重算黄埔系支持（绝对设定），再叠加派系效果
# 复兴社影响力与法西斯主义支持率双向绑定：
#   正向：支持率变化 → 影响力同步（支持率×100）
#   反向：影响力被独立修改（如扶植其他派系挤压）→ 支持率同步（影响力/100）
# 通过记录上一次支持率来区分变化来源，避免循环覆盖。
CHI_sync_fxs_power = {
    # 上一次支持率基线（懒初始化）
    if = {
        limit = { NOT = { has_variable = LA_fxs_pop_last } }
        set_variable = { LA_fxs_pop_last = party_popularity@fascism }
    }
    set_variable = { LA_fxs_pop_now = party_popularity@fascism }
    # 支持率对应的目标影响力（0-100）
    set_variable = { LA_fxs_target = LA_fxs_pop_now }
    multiply_variable = { LA_fxs_target = 100 }

    # 正向：支持率发生了变化 → 用支持率覆盖影响力
    if = {
        limit = { NOT = { check_variable = { var = LA_fxs_pop_now value = LA_fxs_pop_last compare = equals } } }
        set_variable = { LA_power_fxs = LA_fxs_target }
    }
    # 反向：支持率未变，但影响力与支持率不匹配 → 用影响力反向修正支持率
    else = {
        if = {
            limit = { NOT = { check_variable = { var = LA_power_fxs value = LA_fxs_target compare = equals } } }
            set_variable = { LA_fxs_sync_delta = LA_power_fxs }
            subtract_from_variable = { LA_fxs_sync_delta = LA_fxs_target }
            divide_variable = { LA_fxs_sync_delta = 100 }
            add_popularity = { ideology = fascism popularity = LA_fxs_sync_delta }
        }
    }
    # 更新基线（用同步后的实际支持率，保证幂等）
    set_variable = { LA_fxs_pop_last = party_popularity@fascism }
}

    multiply_variable = { CHI_chen_army_exp = LA_hp_scale }
    multiply_variable = { CHI_hu_political_power = LA_hp_scale }
    multiply_variable = { CHI_tang_consumer_goods = LA_hp_scale }
    multiply_variable = { CHI_tang_consumer_goods_display = LA_hp_scale }

    divide_variable = { LA_cc_stability = 100 }

    divide_variable = { LA_fxs_pp_gain = 100 }

    # 生活消费品惩罚（绝对值）= 0.25 × (影响力/100)，最大+25%

    set_variable = { LA_penalty_hp = 0 }
    set_variable = { LA_penalty_fxs = 0 }
    set_variable = { LA_penalty_cz = 0 }
    set_variable = { LA_penalty_zx = 0 }
    set_variable = { LA_penalty_cc_pp = 0 }
    set_variable = { LA_penalty_fxs_pp = 0 }
    set_variable = { LA_penalty_cz_pp = 0 }
    set_variable = { LA_penalty_zx_pp = 0 }
    set_variable = { LA_penalty_pp = 0 }
    set_variable = { LA_penalty_cg = 0 }
    if = {
        limit = { check_variable = { var = LA_power_cc value = 50 compare = greater_than } }
        set_variable = { LA_penalty_cc = 0.25 }
        multiply_variable = { LA_penalty_cc = LA_power_cc }
        divide_variable = { LA_penalty_cc = 100 }
        add_to_variable = { LA_penalty_cg = LA_penalty_cc }
        set_variable = { LA_penalty_cc_pp = LA_penalty_cc }
        add_to_variable = { LA_penalty_pp = LA_penalty_cc_pp }
    }
    if = {
        limit = { check_variable = { var = LA_power_hp value = 50 compare = greater_than } }
        set_variable = { LA_penalty_hp = 0.1 }
        multiply_variable = { LA_penalty_hp = LA_power_hp }
        divide_variable = { LA_penalty_hp = 100 }
    }
    if = {
        limit = { check_variable = { var = LA_power_fxs value = 50 compare = greater_than } }
        set_variable = { LA_penalty_fxs = 0.25 }
        multiply_variable = { LA_penalty_fxs = LA_power_fxs }
        divide_variable = { LA_penalty_fxs = 100 }
        add_to_variable = { LA_penalty_cg = LA_penalty_fxs }
        set_variable = { LA_penalty_fxs_pp = LA_penalty_fxs }
        add_to_variable = { LA_penalty_pp = LA_penalty_fxs_pp }
    }
    if = {
        limit = { check_variable = { var = LA_power_cz value = 50 compare = greater_than } }
        set_variable = { LA_penalty_cz = 0.25 }
        multiply_variable = { LA_penalty_cz = LA_power_cz }
        divide_variable = { LA_penalty_cz = 100 }
        add_to_variable = { LA_penalty_cg = LA_penalty_cz }

        add_to_variable = { LA_penalty_pp = LA_penalty_cz_pp }

        set_variable = { LA_penalty_zx = 0.25 }
        multiply_variable = { LA_penalty_zx = LA_power_zx }
        divide_variable = { LA_penalty_zx = 100 }
        add_to_variable = { LA_penalty_cg = LA_penalty_zx }
        set_variable = { LA_penalty_zx_pp = LA_penalty_zx }
        add_to_variable = { LA_penalty_pp = LA_penalty_zx_pp }
    }

    # 6. 汇总进领袖权威动态修正
    # CC系稳定度（唯一来源，绝对设定保证幂等）
    set_variable = { Leader_Authority_stability_factor = LA_cc_stability }
    # 每日政治点数：国策加成 + 复兴社加成（惩罚已改为百分比系数，见下）
    set_variable = { Leader_Authority_political_power_gain = LA_focus_political_power_gain }
    add_to_variable = { Leader_Authority_political_power_gain = LA_fxs_pp_gain }
    # 政治点数获取（百分比）：派系过大惩罚（池存正数便于显示，写入修正时取负）
    set_variable = { Leader_Authority_political_power_factor = LA_penalty_pp }
    multiply_variable = { Leader_Authority_political_power_factor = -1 }
    # 生活消费品系数：财政系减免 + 派系过大惩罚
    set_variable = { Leader_Authority_consumer_goods_expected_value = LA_cz_cg }
    add_to_variable = { Leader_Authority_consumer_goods_expected_value = LA_penalty_cg }
    # 建造速度：政学系
    set_variable = { Leader_Authority_production_speed_buildings_factor = LA_zx_build }
    # 部队组织度：黄埔系派系过大惩罚（唯一来源，绝对设定保证幂等）
    set_variable = { Leader_Authority_army_org_factor = LA_penalty_hp }

# 扶植CC系：影响力+5%，从复兴社影响力中扣除5%（复兴社影响力下降会同步降低法西斯支持率）
CHI_authority_boost_cc = {
    # 先记录其他三派总和（扣除复兴社前），作为缩放分母
    set_variable = { LA_boost_denom = 100 }
    subtract_from_variable = { LA_boost_denom = LA_power_fxs }
    subtract_from_variable = { LA_boost_denom = LA_power_cc }
    # 从复兴社影响力中扣除 5%（不足则清零）
    if = {
        limit = { check_variable = { var = LA_power_fxs value = 5 compare = greater_than_or_equals } }
        subtract_from_variable = { LA_power_fxs = 5 }
    }
    else = {
        set_variable = { LA_power_fxs = 0 }
    }
    set_variable = { LA_boost_total = 100 }
    subtract_from_variable = { LA_boost_total = LA_power_fxs }
    set_variable = { LA_boost_tmp = LA_power_cc }
    add_to_variable = { LA_boost_tmp = 5 }
    if = {
        limit = { check_variable = { var = LA_boost_tmp value = LA_boost_total compare = greater_than } }
        set_variable = { LA_boost_tmp = LA_boost_total }
    }
    set_variable = { LA_boost_ratio = LA_boost_total }
    subtract_from_variable = { LA_boost_ratio = LA_boost_tmp }
    if = {
        limit = { check_variable = { var = LA_boost_denom value = 0.01 compare = greater_than } }
        divide_variable = { LA_boost_ratio = LA_boost_denom }
    }
    else = {
        set_variable = { LA_boost_ratio = 0 }
    }
    multiply_variable = { LA_power_hp = LA_boost_ratio }
    multiply_variable = { LA_power_cz = LA_boost_ratio }
    multiply_variable = { LA_power_zx = LA_boost_ratio }
    set_variable = { LA_power_cc = LA_boost_tmp }
    # 已在扶植决议内完成五派分配，标记复兴社影响力已同步，避免重算时二次缩放
    set_variable = { LA_fxs_power_last = LA_power_fxs }
    CHI_update_authority_faction_effects = yes
}
