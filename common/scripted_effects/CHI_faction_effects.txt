# 计算派系影响力并更新修正效果
CHI_update_faction_effects = {
   
    # 确保效果变量存在
    if = {
        limit = { NOT = { has_variable = CHI_he_army_strength } }
        set_variable = { CHI_he_army_strength = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_chen_army_exp } }
        set_variable = { CHI_chen_army_exp = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_hu_political_power } }
        set_variable = { CHI_hu_political_power = 0.15 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_tang_consumer_goods } }
        set_variable = { CHI_tang_consumer_goods = -0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_tang_consumer_goods_display } }
        set_variable = { CHI_tang_consumer_goods_display = 0.1 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_hu_power_fuxing } }
        set_variable = { CHI_hu_power_fuxing = 0 }
    }
    
    # 直接设置技能影响力为相应将领的等级（只计算陆军将领）
    set_variable = { CHI_he_power_skill = 0 }
    set_variable = { CHI_chen_power_skill = 0 }
    set_variable = { CHI_hu_power_skill = 0 }
    set_variable = { CHI_tang_power_skill = 0 }
    
    # 计算陆军将领的技能影响力
    every_army_leader = {
        limit = { AND = { has_trait = heyingqinxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_he_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = chenchengxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_chen_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = huzongnanxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_hu_power_skill = skill }
    }
    every_army_leader = {
        limit = { AND = { has_trait = tangenboxi is_employed_by = ROOT } }
        add_to_variable = { PREV.CHI_tang_power_skill = skill }
    }
    
    # 计算总影响力 = 基础影响力 + 内阁加成 + 技能影响力
    set_variable = { CHI_he_power = CHI_he_power_base }
    add_to_variable = { CHI_he_power = CHI_he_power_skill }
    
    set_variable = { CHI_chen_power = CHI_chen_power_base }
    add_to_variable = { CHI_chen_power = CHI_chen_power_skill }
    
    set_variable = { CHI_hu_power = CHI_hu_power_base }
    add_to_variable = { CHI_hu_power = CHI_hu_power_skill }
    add_to_variable = { CHI_hu_power = CHI_hu_power_fuxing }
    
    set_variable = { CHI_tang_power = CHI_tang_power_base }
    add_to_variable = { CHI_tang_power = CHI_tang_power_skill }
    
    # 计算总派系影响力
    set_variable = { CHI_total_clique_power = 0 }
    add_to_variable = { CHI_total_clique_power = CHI_he_power }
    add_to_variable = { CHI_total_clique_power = CHI_chen_power }
    add_to_variable = { CHI_total_clique_power = CHI_hu_power }
    add_to_variable = { CHI_total_clique_power = CHI_tang_power }
    
    # 确保总派系影响力变量存在
    if = {
        limit = { NOT = { has_variable = CHI_total_clique_power } }
        set_variable = { CHI_total_clique_power = 0 }
    }
    
    # 确保影响力百分比变量存在
    if = {
        limit = { NOT = { has_variable = CHI_he_power_percent } }
        set_variable = { CHI_he_power_percent = 25 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_chen_power_percent } }
        set_variable = { CHI_chen_power_percent = 25 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_hu_power_percent } }
        set_variable = { CHI_hu_power_percent = 25 }
    }
    if = {
        limit = { NOT = { has_variable = CHI_tang_power_percent } }
        set_variable = { CHI_tang_power_percent = 25 }
    }
    
    # 计算影响力百分比
    if = {
        limit = { CHI_total_clique_power > 0 }
        set_variable = { CHI_he_power_percent = CHI_he_power }
        divide_variable = { CHI_he_power_percent = CHI_total_clique_power }
        multiply_variable = { CHI_he_power_percent = 100 }
        
        set_variable = { CHI_chen_power_percent = CHI_chen_power }
        divide_variable = { CHI_chen_power_percent = CHI_total_clique_power }
        multiply_variable = { CHI_chen_power_percent = 100 }
        
        set_variable = { CHI_hu_power_percent = CHI_hu_power }
        divide_variable = { CHI_hu_power_percent = CHI_total_clique_power }
        multiply_variable = { CHI_hu_power_percent = 100 }
        
        set_variable = { CHI_tang_power_percent = CHI_tang_power }
        divide_variable = { CHI_tang_power_percent = CHI_total_clique_power }
        multiply_variable = { CHI_tang_power_percent = 100 }
    }
    else {
        set_variable = { CHI_he_power_percent = 25 }
        set_variable = { CHI_chen_power_percent = 25 }
        set_variable = { CHI_hu_power_percent = 25 }
        set_variable = { CHI_tang_power_percent = 25 }
    }
    
    # 应用派系影响力效果
    # 直接根据影响力百分比计算变量值（仅保留正面属性）

    # 何应钦派系
    # 部队HP: 0.15 * (影响力百分比/100)
    set_variable = { CHI_he_army_strength = 0.15 }
    multiply_variable = { CHI_he_army_strength = CHI_he_power_percent }
    divide_variable = { CHI_he_army_strength = 100 }

    # 陈诚派系
    # 陆军经验获取因子: 0.15 * (影响力百分比/100)
    set_variable = { CHI_chen_army_exp = 0.15 }
    multiply_variable = { CHI_chen_army_exp = CHI_chen_power_percent }
    divide_variable = { CHI_chen_army_exp = 100 }

    # 胡宗南派系
    # 政治点数因子: 0.15 * (影响力百分比/100)
    set_variable = { CHI_hu_political_power = 0.15 }
    multiply_variable = { CHI_hu_political_power = CHI_hu_power_percent }
    divide_variable = { CHI_hu_political_power = 100 }

    # 汤恩伯派系
    # 生活消费品系数（实际修正值，负数）: -0.15 * (影响力百分比/100)
    set_variable = { CHI_tang_consumer_goods = -0.15 }
    multiply_variable = { CHI_tang_consumer_goods = CHI_tang_power_percent }
    divide_variable = { CHI_tang_consumer_goods = 100 }
    # 生活消费品系数（显示用，正数）: 0.15 * (影响力百分比/100)
    set_variable = { CHI_tang_consumer_goods_display = 0.15 }
    multiply_variable = { CHI_tang_consumer_goods_display = CHI_tang_power_percent }
    divide_variable = { CHI_tang_consumer_goods_display = 100 }

}

# 计算领袖权威：五人顾问（胡宗南/戴笠/康泽/贺衷寒/邓文仪）在任加成
# 设计为幂等的一次性触发：顾问激活/替换只会使效果变化一次，重算结果恒定，绝不逐日累积
CHI_update_leader_authority_effect = {
    # 统计当前在任的五人顾问数量
    set_variable = { LA_advisor_count = 0 }
    if = {
        limit = { has_idea = CHI_HuZongNan }
        add_to_variable = { LA_advisor_count = 1 }
    }
    if = {
        limit = { has_idea = CHI_DaiLi_P }
        add_to_variable = { LA_advisor_count = 1 }
    }
    if = {
        limit = { has_idea = CHI_KangZe_P }
        add_to_variable = { LA_advisor_count = 1 }
    }
    if = {
        limit = { has_idea = CHI_HeZhongHan_A }
        add_to_variable = { LA_advisor_count = 1 }
    }
    if = {
        limit = { has_idea = CHI_DengWenYi_P }
        add_to_variable = { LA_advisor_count = 1 }
    }

    # —— 总值 = 国策基础(LA_focus_*) + 0.025×人数 ——
    # 绝对设定：无论重算多少次结果恒定，旧存档中的错误累积一天内被自动覆盖修正
    set_variable = { Leader_Authority_political_power_factor = LA_focus_political_power_factor }
    set_variable = { LA_contrib_pp = 0.025 }
    multiply_variable = { LA_contrib_pp = LA_advisor_count }
    add_to_variable = { Leader_Authority_political_power_factor = LA_contrib_pp }

    set_variable = { Leader_Authority_stability_factor = LA_focus_stability_factor }
    set_variable = { LA_contrib_stab = 0.025 }
    multiply_variable = { LA_contrib_stab = LA_advisor_count }
    add_to_variable = { Leader_Authority_stability_factor = LA_contrib_stab }

    set_variable = { Leader_Authority_war_support_factor = LA_focus_war_support_factor }
    set_variable = { LA_contrib_ws = 0.025 }
    multiply_variable = { LA_contrib_ws = LA_advisor_count }
    add_to_variable = { Leader_Authority_war_support_factor = LA_contrib_ws }

    # —— 每日法西斯主义支持率：3-4人 +0.05；5人 +0.1；其余为0（绝对设定，无累积）——
    if = {
        limit = { check_variable = { var = LA_advisor_count value = 5 compare = greater_than_or_equals } }
        set_variable = { Leader_Authority_fascism_drift = 0.1 }
    }
    else_if = {
        limit = { check_variable = { var = LA_advisor_count value = 3 compare = greater_than_or_equals } }
        set_variable = { Leader_Authority_fascism_drift = 0.05 }
    }
    else = {
        set_variable = { Leader_Authority_fascism_drift = 0 }
    }

    # 记录当前人数（仅供调试查看）
    set_variable = { LA_advisor_count_prev = LA_advisor_count }
}
