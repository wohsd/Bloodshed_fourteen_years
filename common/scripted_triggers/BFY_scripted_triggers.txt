		}
		is_core_of = MEG
	}
}

# BFY补充：CHI经济脚本（CHI_scripted_effects.txt）引用但原本缺失定义的自定义触发器
# 国家拥有任一“工业化”动态修正（1-4级）

	OR = {
		has_dynamic_modifier = { modifier = CHI_industrialisation_dynamic_modifier_1 }
		has_dynamic_modifier = { modifier = CHI_industrialisation_dynamic_modifier_2 }
		has_dynamic_modifier = { modifier = CHI_industrialisation_dynamic_modifier_3 }
		has_dynamic_modifier = { modifier = CHI_industrialisation_dynamic_modifier_4 }
	}
}

# 官僚制2级/3级动态修正已无任何实际效果（全部加成变量约等于0），应直接移除而非升降级
# 判定阈值与 swap_or_remove_bureaucracy 各转换分支中的空壳判定保持一致
need_to_remove_bureaucracy = {
	OR = {
		AND = {
			has_dynamic_modifier = { modifier = CHI_bureaucracy_dynamic_modifier_2 }
			check_variable = { CHI_bureaucracy_political_power_gain > -0.01 }
			check_variable = { CHI_bureaucracy_mobilization_laws_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_trade_laws_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_economy_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_political_advisor_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_tank_manufacturer_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_naval_manufacturer_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_aircraft_manufacturer_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_materiel_manufacturer_cost_factor < 0.01 }
			check_variable = { CHI_bureaucracy_industrial_concern_cost_factor < 0.01 }
		}
		AND = {
			has_dynamic_modifier = { modifier = CHI_bureaucracy_dynamic_modifier_3 }
			check_variable = { CHI_bureaucracy_stability_factor > -0.11 }
			check_variable = { CHI_bureaucracy_consumer_goods_factor < 0.01 }
		}
	}
}
