on_actions = {
    on_startup = {
        effect = {
            CHI = {
                # 初始化并计算派系影响力
                CHI_update_faction_effects = yes
                # 添加动态修正
                add_dynamic_modifier = {
                    modifier = Republic_of_China_Military_Academy
                }
            }
        }
    }
    
    # 每日更新派系影响力
    on_daily = {
        effect = {
            CHI = {
                CHI_update_faction_effects = yes
            }
        }
    }
    
}
