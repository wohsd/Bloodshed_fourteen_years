on_actions = {
    on_startup = {
        effect = {
            CHI = {
                # 初始化并计算派系影响力、领袖权威与五大派系（链式调用）
                CHI_update_authority_faction_effects = yes
                # 添加动态修正
                add_dynamic_modifier = {
                    modifier = Republic_of_China_Military_Academy
                }
            }
        }
    }

    # 每日更新派系影响力与领袖权威
    on_daily = {
        effect = {
            CHI = {
                # 每日重算（顾问增减、决议扶植后自动更新）
                CHI_update_authority_faction_effects = yes
            }
        }
    }

}
