on_actions = {
    on_startup = {
        effect = {
            CHI = {
                # 初始化并计算派系影响力
                CHI_update_faction_effects = yes
            }
        }
    }
    
    # 每日更新派系影响力
    on_daily = {
        effect = {
            CHI = {
                CHI_update_faction_effects = yes
            }
        }
    }
    
    # 将领升级时更新派系影响力
    on_unit_leader_level_up = {
        effect = {
            CHI = {
                CHI_update_faction_effects = yes
            }
        }
    }
}
