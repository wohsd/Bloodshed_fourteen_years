CAR_beiping_gamble = {

	# 会师北平城下 —— 使命型决议（仿红军长征 Long_march_mission 模式）
	# activation 自动开始；30日内达成 available(攻占北平)=complete_effect 成功
	# 逾期=timeout_effect 失败，全军由东北军收编
	CAR_storm_beiping = {
		icon = military_operation
		visible = {
			NOT = { controls_state = 74 }
		}
		activation = {
		} 
		available = {
			controls_state = 74
		}
		days_mission_timeout = 30
		fixed_random_seed = no
		fire_only_once = yes
		is_good = no

		# 成功：如期光复北平
		complete_effect = {
			add_war_support = 0.05
			add_stability = 0.05
			add_political_power = 50
			set_country_flag = CAR_beiping_secured
		}

		# 失败：逾期未能攻下北平，同盟军瓦解
		timeout_effect = {
			if = {
				limit = { ROD = { exists = yes } }
				ROD = { annex_country = { target = CAR } }
			}
			if = {
				limit = {
					NOT = { ROD = { exists = yes } }
					CHI = { exists = yes }
				}
				CHI = { annex_country = { target = CAR } }
			}
		}
	}
}
