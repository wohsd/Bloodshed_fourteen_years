# 1931年12月15日：宁粤对峙，蒋通电下野
country_event = {
	id = CHIJiang.1
	title = CHIJiang.1.t
	desc = CHIJiang.1.d
	picture = GFX_report_event_generic_sign_treaty1

	trigger = {
		original_tag = CHI
		date > 1931.12.14
		# 东北失守：48 / 66 / 67 / 75 已被关东军占领（由 Hidden_News.1 九一八事变转入）
		48 = { is_controlled_by = JAP }
		66 = { is_controlled_by = JAP }
		67 = { is_controlled_by = JAP }
		75 = { is_controlled_by = JAP }
		NOT = { has_country_flag = CHI_jiang_stage_1931 }
	}
	mean_time_to_happen = { days = 2 }
	fire_only_once = yes

	immediate = {
		set_country_flag = CHI_jiang_stage_1931
	}

	option = {
		name = CHIJiang.1.a
		ai_chance = { base = 85 }

		set_country_flag = CHI_jiang_resigned_1931
		add_political_power = 25
		add_stability = -0.05
		add_war_support = 0.03

		# 林森继任国民政府主席（名义元首，实权仍在蒋系军中）
		# 林森在 characters 文件中已持有 despotism 领袖角色，只能用 promote_character 提升
		promote_character = {
			character = CHI_LingSeng
			ideology = despotism
		}
	}

	#option = {
	#	name = CHIJiang.1.b
	#	ai_chance = { base = 15 }

	#	set_country_flag = CHI_jiang_defied_1931
	#	add_political_power = -40
	#	add_stability = -0.10
	#}
}
