# 蒋介石开封诱捕、军法处决韩复榘（CHI决策）
country_event = {
	id = ROA_1937.2
	title = ROA_1937.2.t
	desc = ROA_1937.2.d
	fire_only_once = yes
	#picture = GFX_ROG_1931_1
	is_triggered_only = yes
    immediate = {
    }
	# 历史主线：军法处决，随后中央借势整编鲁省军政
	option = {
		name = ROA_1937.2.a
		add_political_power = 50
		add_war_support = 0.05
		if = {
			limit = {
				ROA = { exists = yes }
			}
			ROA = {
				# 韩复榘以“违抗命令，擅自撤退”罪被枪决
				retire_country_leader = {
					ideology = despotism
					kill = yes
				}
				# 沈鸿烈接任山东省政，孙桐萱接掌第三集团军
				promote_character = {
					character = ROA_ShengHongLie
					ideology = despotism
				}
				set_country_flag = ROA_HanFuQu_executed
				add_stability = -0.10
				add_war_support = -0.05
			}
			country_event = { id = ROA_1937.3 days = 10 }
		}
	}
	# 支线：念其中原大战旧勋，仅撤职而不问斩，山东维持半独立
	option = {
		name = ROA_1937.2.b
		add_political_power = -25
		if = {
			limit = {
				ROA = { exists = yes }
			}
			ROA = {
				retire_country_leader = {
					ideology = despotism
				}
				promote_character = {
					character = ROA_ShengHongLie
					ideology = despotism
				}
				set_country_flag = ROA_HanFuQu_spared
				add_stability = -0.05
			}
		}
	}
}

# 整编鲁省军政：CHI决定彻底中央化还是暂留第三集团军建制
country_event = {
	id = ROA_1937.3
	title = ROA_1937.3.t
	desc = ROA_1937.3.d
	fire_only_once = yes
	is_triggered_only = yes
    immediate = {
    }
	# 全面整编：第三集团军通电归顺，山东军政悉归中央（CHI吞并ROA）
	option = {
		name = ROA_1937.3.a
		hidden_effect = {
			if = {
				limit = {
					ROA = { exists = yes }
				}
				ROA = {
					set_country_flag = ROA_central_army_takeover
					country_event = { id = ROA_1937.4 days = 3 }
				}
			}
		}
	}
	# 暂缓中央化：沈鸿烈主鲁、孙桐萱掌军，ROA成为CHI傀儡
	option = {
		name = ROA_1937.3.b
		add_stability = 0.05
		hidden_effect = {
			if = {
				limit = {
					ROA = {
						exists = yes
						NOT = { is_subject_of = CHI }
					}
				}
				ROA = {
					puppet = CHI
					set_country_flag = ROA_third_army_retained
					add_stability = 0.05
				}
			}
		}
	}
}
