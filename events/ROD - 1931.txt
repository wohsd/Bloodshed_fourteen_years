# ==============================================
# 九一八事变后南京政府态度：抵抗 or 不抵抗
# 决定 ROD_focus_0（听命中央）vs ROD_focus_1（联共抗日）国策树走向
# 由 Hidden_News.1（九一八事变）的 ROD scope 于 +3 天调度
# ==============================================
country_event = {
	id = ROD1931.50
	title = ROD1931.50.t
	desc = ROD1931.50.d
	picture = GFX_ROD_1931_50

	fire_only_once = yes
	is_triggered_only = yes

	# 必须在九一八后，且两个 flag 都没设过（防万一）
	trigger = {
		date > 1931.9.18
		# 哈尔滨/长春/吉林/沈阳 全部被日军占领才触发（有一城在手就不触发）
		NOT = {
			any_state = {
				OR = {
					state = 24	# 哈尔滨
					state = 36	# 长春
					state = 40	# 吉林
					state = 50	# 沈阳
				}
				NOT = { is_controlled_by = JAP }
			}
		}
	}

	option = {
		name = ROD1931.50.a	# 遵命执行，不抵抗
		complete_national_focus = ROD_focus_0
		hidden_effect = {
			set_country_flag = ROD_918_obey
			add_stability = 0.02
		}
	}
	option = {
		name = ROD1931.50.b	# 通电全国，奋起抗日
		complete_national_focus = ROD_focus_1
		hidden_effect = {
			set_country_flag = ROD_918_resist
			add_war_support = 0.05
			add_political_power = 10
		}
	}
}
