# 压力巡检（隐藏）：每次压力变动后2天自查，挂/摘阶梯debuff
country_event = {
	id = ROD1931.990
	hidden = yes
	is_triggered_only = yes

	immediate = {
		# ≥90：严重状态
		if = {
			limit = {
				check_variable = { ROD_zhang_xueliang_stress > 89 }
				NOT = { has_idea = ROD_zhang_stress_severe }
			}
			add_ideas = ROD_zhang_stress_severe
			country_event = { id = ROD1931.992 days = 1 }
		}
		else_if = {
			limit = {
				has_idea = ROD_zhang_stress_severe
				check_variable = { ROD_zhang_xueliang_stress < 90 }
			}
			remove_ideas = ROD_zhang_stress_severe
		}
		# ≥70：积劳成疾
		if = {
			limit = {
				check_variable = { ROD_zhang_xueliang_stress > 69 }
				NOT = { has_idea = ROD_zhang_stress_high }
			}
			add_ideas = ROD_zhang_stress_high
			country_event = { id = ROD1931.991 days = 1 }
		}
		else_if = {
			limit = {
				has_idea = ROD_zhang_stress_high
				check_variable = { ROD_zhang_xueliang_stress < 70 }
			}
			remove_ideas = ROD_zhang_stress_high
		}
	}
}
