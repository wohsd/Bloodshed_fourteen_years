# 革命的胜利
# 临委会完成「第三路线」起义后击败国民政府（CHI投降，见 on_capitulation）
# · 移除「临委会起义」动态修正，获得「内战创伤」动态修正
# · 解锁「重建工作」决议组
# · 自动完成战后三议题国策：军队战斗力问题 / 解决军队开支 / 新训政
# 内战创伤三类等级基础均为4；战时稳健国策各-1级，战时激进国策各+1级
# ============================================================
country_event = {
	id = RFC_country.6
	title = RFC_country.6.t
	desc = RFC_country.6.d
	picture = GFX_report_event_generic_signed_treaty
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# 临委会起义动员状态解除
			remove_dynamic_modifier = { modifier = Provisional_Committee_Uprising }
			clr_country_flag = RFC_pcq_war_engaged
			set_country_flag = RFC_victory_won

			# 三类内战创伤基础等级均为4
			set_variable = { RFC_cw_level_army = 4 }
			set_variable = { RFC_cw_level_industry = 4 }
			set_variable = { RFC_cw_level_govt = 4 }

			# 稳健的战时国策：减轻对应创伤1级
			if = {
				limit = { has_completed_focus = Military_Crash_Course }
				add_to_variable = { RFC_cw_level_army = -1 }
			}
			if = {
				limit = { has_completed_focus = Production_Self_Help_Movement }
				add_to_variable = { RFC_cw_level_industry = -1 }
			}
			if = {
				limit = { has_completed_focus = Aid_refugees }
				add_to_variable = { RFC_cw_level_govt = -1 }
			}

			# 激进的战时国策：加重对应创伤1级
			if = {
				limit = { has_completed_focus = Suicide_Squad }
				add_to_variable = { RFC_cw_level_army = 1 }
			}
			if = {
				limit = { has_completed_focus = Requisition_factory }
				add_to_variable = { RFC_cw_level_industry = 1 }
			}
			if = {
				limit = { has_completed_focus = Forced_labor }
				add_to_variable = { RFC_cw_level_govt = 1 }
			}

			# 等级换算为修正数值（每级：军事-3%组织度/-3%恢复；工业-5%建造/-3%产出；机构-4%PP/-3%稳定）
			set_variable = { RFC_cw_army = RFC_cw_level_army }
			multiply_variable = { RFC_cw_army = -0.03 }
			set_variable = { RFC_cw_army_morale = RFC_cw_level_army }
			multiply_variable = { RFC_cw_army_morale = -0.03 }

			set_variable = { RFC_cw_build = RFC_cw_level_industry }
			multiply_variable = { RFC_cw_build = -0.05 }
			set_variable = { RFC_cw_factory = RFC_cw_level_industry }
			multiply_variable = { RFC_cw_factory = -0.03 }

			set_variable = { RFC_cw_pp = RFC_cw_level_govt }
			multiply_variable = { RFC_cw_pp = -0.04 }
			set_variable = { RFC_cw_stability = RFC_cw_level_govt }
			multiply_variable = { RFC_cw_stability = -0.03 }

			# 获得「内战创伤」并解锁「重建工作」决议组
			add_dynamic_modifier = { modifier = Civil_War_Trauma }
			set_country_flag = RFC_civil_war_trauma

			# 自动完成战后三议题国策（玩家无法手动点选）
			complete_national_focus = new_focus_18
			complete_national_focus = new_focus_1
			complete_national_focus = new_focus_19
		}
	}

	option = {
		name = RFC_country.6.a
	}
}
