			RFC = {

				remove_dynamic_modifier = { modifier = Three_Major_Policies_Nominal }
				add_dynamic_modifier = { modifier = Three_Major_Policies_Initial }
				# 追加效果
				add_to_variable = { RFC_three_policies_pp = 0.05 }
				add_to_variable = { RFC_three_policies_war_support = 0.05 }
				add_stability = 0.05
			}
		}
	}

	option = {
		name = RFC_NEW.2.b
		# 我们不能相信他们

		# RFC共产主义支持率-10%

			add_opinion_modifier = { target = RFC modifier = RFC_second_united_front_refused }
			RFC = {
				add_opinion_modifier = { target = ROOT modifier = RFC_second_united_front_refused }
				add_popularity = { ideology = communism popularity = -0.10 }
			}
		}
	}
