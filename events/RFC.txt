# 民主同盟的胜利（新闻次日由RFC收到，仅触发一次）
# · 移除「临委会起义」动员修正，获得非常强力的「从内战中恢复」debuff
# · 解锁战后恢复国策支线（收拾残局），通过国策逐级削弱debuff
# · 自动完成战后三议题国策：军队战斗力问题 / 解决军队开支 / 新训政
# 三类创伤基础均为5级；战时稳健国策各-1级，战时激进国策各+1级
# ============================================================
country_event = {
	id = RFC_country.6
	title = RFC_country.6.t
	desc = RFC_country.6.d
	picture = GFX_report_event_generic_signed_treaty
	fire_only_once = yes
	is_triggered_only = yes

	immediate = {
		hidden_effect = {
			# 临委会起义动员状态解除
			remove_dynamic_modifier = { modifier = Provisional_Committee_Uprising }
			clr_country_flag = RFC_pcq_war_engaged
			set_country_flag = RFC_victory_won

			# 三类内战创伤基础等级均为5
			set_variable = { RFC_rc_level_army = 5 }
			set_variable = { RFC_rc_level_city = 5 }
			set_variable = { RFC_rc_level_pol = 5 }

			# 稳健的战时国策：减轻对应创伤1级
			if = {
				limit = { has_completed_focus = Military_Crash_Course }
				add_to_variable = { RFC_rc_level_army = -1 }
			}
			if = {
				limit = { has_completed_focus = Production_Self_Help_Movement }
				add_to_variable = { RFC_rc_level_city = -1 }
			}
			if = {
				limit = { has_completed_focus = Aid_refugees }
				add_to_variable = { RFC_rc_level_pol = -1 }
			}

			# 激进的战时国策：加重对应创伤1级
			if = {
				limit = { has_completed_focus = Suicide_Squad }
				add_to_variable = { RFC_rc_level_army = 1 }
			}
			if = {
				limit = { has_completed_focus = Requisition_factory }
				add_to_variable = { RFC_rc_level_city = 1 }
			}
			if = {
				limit = { has_completed_focus = Forced_labor }
				add_to_variable = { RFC_rc_level_pol = 1 }
			}

			# 城市破坏：每级 建造速度-6%、工厂产出-4%
			set_variable = { RFC_rc_build = RFC_rc_level_city }
			multiply_variable = { RFC_rc_build = -0.06 }
			set_variable = { RFC_rc_factory = RFC_rc_level_city }
			multiply_variable = { RFC_rc_factory = -0.04 }

			# 政治激化：每级 政治点数获取-6%、稳定度-4%
			set_variable = { RFC_rc_pp = RFC_rc_level_pol }
			multiply_variable = { RFC_rc_pp = -0.06 }
			set_variable = { RFC_rc_stab = RFC_rc_level_pol }
			multiply_variable = { RFC_rc_stab = -0.04 }

			# 军队混乱：每级 陆军组织度-4%、组织度恢复-4%
			set_variable = { RFC_rc_org = RFC_rc_level_army }
			multiply_variable = { RFC_rc_org = -0.04 }
			set_variable = { RFC_rc_morale = RFC_rc_level_army }
			multiply_variable = { RFC_rc_morale = -0.04 }

			# 获得「从内战中恢复」（战后恢复国策逐级削弱，终策「百废俱兴」移除）
			add_dynamic_modifier = { modifier = RFC_Recovering_From_Civil_War }

			# 自动完成战后三议题国策（玩家无法手动点选）
			complete_national_focus = new_focus_18
			complete_national_focus = new_focus_1
			complete_national_focus = new_focus_19
		}
	}

	option = {
		name = RFC_country.6.a
	}
}
