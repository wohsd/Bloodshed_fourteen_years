# 派系影响力事件

country_event = {
    id = CHI_faction.1
    title = "CHI_faction.1.title"
    desc = "CHI_faction.1.desc"
    picture = GFX_idea_Republic_of_China_Military_Academy
    
    is_triggered_only = yes
    
    trigger = {
        tag = CHI
    }
    
    immediate = {
        # 检查何应钦派系影响力
        if = {
            limit = { check_variable = { CHI_he_power_percent > 50 } }
            set_variable = { CHI_he_supply_consumption = -0.15 }
            set_variable = { CHI_he_supply_range = 0.15 }
            set_variable = { CHI_he_planning_speed = -0.15 }
        }
        else_if = {
            limit = { check_variable = { CHI_he_power_percent >= 25 } }
            set_variable = { CHI_he_supply_consumption = -0.1 }
            set_variable = { CHI_he_supply_range = 0.1 }
            set_variable = { CHI_he_planning_speed = -0.1 }
        }
        else {
            set_variable = { CHI_he_supply_consumption = -0.05 }
            set_variable = { CHI_he_supply_range = 0.05 }
            set_variable = { CHI_he_planning_speed = -0.05 }
        }
        
        # 检查陈诚派系影响力
        if = {
            limit = { check_variable = { CHI_chen_power_percent > 50 } }
            set_variable = { CHI_chen_army_exp = 0.15 }
            set_variable = { CHI_chen_political_power = -0.1 }
        }
        else_if = {
            limit = { check_variable = { CHI_chen_power_percent >= 25 } }
            set_variable = { CHI_chen_army_exp = 0.1 }
            set_variable = { CHI_chen_political_power = -0.05 }
        }
        else {
            set_variable = { CHI_chen_army_exp = 0.05 }
            set_variable = { CHI_chen_political_power = -0.02 }
        }
        
        # 检查胡宗南派系影响力
        if = {
            limit = { check_variable = { CHI_hu_power_percent > 50 } }
            set_variable = { CHI_hu_army_org = 1.5 }
            set_variable = { CHI_hu_commander_exp = -0.15 }
        }
        else_if = {
            limit = { check_variable = { CHI_hu_power_percent >= 25 } }
            set_variable = { CHI_hu_army_org = 1.0 }
            set_variable = { CHI_hu_commander_exp = -0.1 }
        }
        else {
            set_variable = { CHI_hu_army_org = 0.5 }
            set_variable = { CHI_hu_commander_exp = -0.05 }
        }
        
        # 检查汤恩伯派系影响力
        if = {
            limit = { check_variable = { CHI_tang_power_percent > 50 } }
            set_variable = { CHI_tang_industrial = -0.15 }
            set_variable = { CHI_tang_irregular_attack = 0.15 }
            set_variable = { CHI_tang_irregular_defence = 0.15 }
            set_variable = { CHI_tang_irregular_org = 0.15 }
        }
        else_if = {
            limit = { check_variable = { CHI_tang_power_percent >= 25 } }
            set_variable = { CHI_tang_industrial = -0.1 }
            set_variable = { CHI_tang_irregular_attack = 0.1 }
            set_variable = { CHI_tang_irregular_defence = 0.1 }
            set_variable = { CHI_tang_irregular_org = 0.1 }
        }
        else {
            set_variable = { CHI_tang_industrial = -0.05 }
            set_variable = { CHI_tang_irregular_attack = 0.05 }
            set_variable = { CHI_tang_irregular_defence = 0.05 }
            set_variable = { CHI_tang_irregular_org = 0.05 }
        }
    }
    
    option = {
        name = "CHI_faction.1.option"
    }
}
