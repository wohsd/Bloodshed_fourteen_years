# 权力平衡事件
country_event = {
	id = CHI_struggle_balance_left_high
	title = CHI_struggle_balance_left_high.t
	desc = CHI_struggle_balance_left_high.d
	picture = generic_political_crisis

	is_triggered_only = yes

	option = {
		name = CHI_struggle_balance_left_high.a
		CHI = {
			# 减少更多政治点数，增加更多生活消费品，每日增加更多中立主义支持率
			add_to_variable = {
				ZHRS_political_power_gain = -0.2
			}
			add_to_variable = {
				ZHRS_consumer_goods = 0.05
			}
			add_to_variable = {
				ZHRS_neutrality_drift = 0.01
			}
			# 政治平衡每月额外向左侧移动很多
			add_power_balance_value = {
				id = CHI_struggle_balance
				value = -0.05
			}
		}
	}
}

	option = {
		name = CHI_struggle_balance_mid.a
		CHI = {
			# 动态修正"领袖权威"额外增加5%政治点数、战争支持率、稳定性
			add_to_variable = {
				ZHRS_political_power_gain = 0.1
			}
			add_to_variable = {
				ZHRS_war_support = 0.05
			}

				ZHRS_stability = 0.05
			}
			# 当党政平衡时，把CHI_hu_power_fuxing设置为零
			set_variable = {
				CHI_hu_power_fuxing = 0
			}
		}
	}
}
