	id = PRC_political.3
	title = PRC_political.3.t
	desc = PRC_political.3.d
	picture = GFX_anti_japanese_speech
	
	fire_only_once = yes
	
	trigger = {
		tag = PRC
		has_country_flag = PRC_focus85
		OR = {
			AND = {
				power_balance_value = {
					id = PRC_struggle_balance
					value < -0.8
				}
				has_country_flag = PRC_Mao_FLAG
			}
			AND = {
				power_balance_value = {
					id = PRC_struggle_balance
					value > 0.8
				}
				has_country_flag = PRC_Wang_FLAG
			}
		}
	}
	
	mean_time_to_happen = {
		days = 20
	}
	
	option = {
		name = PRC_political.3.a
		add_stability = -0.1
		add_war_support = -0.05
		add_power_balance_value = {
			id = PRC_struggle_balance
			value = -0.1
		}
	}
}
