# 1931年12月10日：国联理事会决议派遣调查团
country_event = {
	id = ROD_lytton.1
	title = ROD_lytton.1.t
	desc = ROD_lytton.1.d
	picture = GFX_ROD_NEA_2

	fire_only_once = yes
	trigger = {
		original_tag = ROD
		date > 1931.12.09
		OR = {
			has_country_flag = ROD_918_obey
			has_country_flag = ROD_918_resist
		}
		NOT = { has_country_flag = ROD_lytton_commission }
	}
	mean_time_to_happen = { days = 3 }

	option = {
		name = ROD_lytton.1.a
		add_political_power = 20
		add_war_support = 0.03
		set_country_flag = ROD_lytton_commission
		hidden_effect = {
			news_event = { id = ROD_lytton.10 }
		}
	}
}
